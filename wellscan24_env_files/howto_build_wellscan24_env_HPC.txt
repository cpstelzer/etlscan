# Build on Leo5

module load Anaconda3/2023.10/miniconda-base-2023.10
eval "$($UIBK_CONDA_DIR/bin/conda shell.bash hook)"

# Create the environment from high-level packages
mamba create -y -p wellscan_env -c conda-forge \
  python=3.9 \
  numpy scipy pandas matplotlib \
  opencv \
  pyqt \
  scikit-image scikit-learn \
  pyyaml networkx joblib threadpoolctl \
  trackpy pyefd \
  pip

# Activate the environment
conda activate $SCRATCH/wellscan_env

# Explicitly install pytorch after activating new conda environment
mamba install -y -c pytorch pytorch torchvision torchaudio cpuonly
# (GPU support is also possible, but requires cluster-specific command!)

# Install the pip-only packages
python -m pip install --no-user -r $SCRATCH/env_setup/pip-requirements.txt
